#!/bin/bash

echo "ubuntu:ubuntu" | sudo chpasswd

sudo sed -i 's/^[#[:space:]]*PasswordAuthentication.*/PasswordAuthentication yes/' /etc/ssh/sshd_config

echo "PasswordAuthentication yes" | sudo tee /etc/ssh/sshd_config.d/00-password-auth.conf

sudo sshd -t && sudo systemctl restart ssh

echo "Ubuntu password set to: ubuntu"
